      days = 90

    }

  bucket = aws_s3_bucket.uploads.id
  
  lambda_function {
    lambda_function_arn = aws_lambda_function.document_processor.arn
    events              = ["s3:ObjectCreated:*"]
    filter_prefix       = "uploads/"
    filter_suffix       = ".pdf"
  }
  
  lambda_function {
    lambda_function_arn = aws_lambda_function.document_processor.arn
    events              = ["s3:ObjectCreated:*"]
    filter_prefix       = "uploads/"
    filter_suffix       = ".xlsx"
  }
}

resource "aws_lambda_permission" "allow_s3" {
  statement_id  = "AllowExecutionFromS3"
  action        = "lambda:InvokeFunction"
  function_name = aws_lambda_function.document_processor.function_name
  principal     = "s3.amazonaws.com"
  source_arn    = aws_s3_bucket.uploads.arn
}

output "upload_bucket_name" {
  value = aws_s3_bucket.uploads.id
}

output "document_processor_function_name" {

## 🎯 **KEY FEATURES ADDED**

### ✅ **What This Implementation Provides:**

1. **PDF/Excel Upload** ✓
   - Multi-format support (PDF, XLS, XLSX)
   - 50MB file size limit
   - User authentication required

2. **AWS Textract Integration** ✓
   - Automatic text extraction from PDFs
   - Table detection (for Excel-like data in PDFs)
   - Multi-page document handling

3. **Async Processing** ✓
   - Non-blocking upload (returns immediately)
   - Background processing via Lambda
   - Real-time status updates via polling

4. **Intelligent Chunking** ✓
   - 512-token chunks with 50-token overlap
   - Preserves context across chunks
   - Metadata tagging for filtering

5. **Vector Indexing** ✓
   - Automatic embedding generation
   - OpenSearch indexing
   - Semantic search enabled

6. **RAG Integration** ✓
   - Uploaded documents automatically available for RAG
   - No code changes needed - existing chat uses them
   - Source citation shows document origin

7. **Progress Tracking** ✓
   - 0-100% progress indicator
   - Status: pending → processing → completed/failed
   - Error handling with messages

8. **Security** ✓
   - User-scoped documents (can't see others' uploads)
   - S3 encryption at rest
   - Cognito authentication required
   - Document expiry after 90 days
